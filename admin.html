/** * 오더문의 백엔드 v2 (Google Apps Script) * - 접수 저장 (index.html 이 사용) * - 접수 목록 읽기 / 수정 / 삭제 (admin.html 이 사용) * * ─── 기존 v1 을 쓰고 계셨다면 ─────────────────────────────── * Apps Script 편집기에서 기존 코드를 전부 지우고 이 내용으로 교체한 뒤, * [배포] → [배포 관리] → 연필 아이콘 → 버전 [새 버전] → [배포] * ※ '새 배포'가 아니라 '배포 관리'로 해야 기존 URL이 유지됩니다. * ────────────────────────────────────────────────────────── */ const NOTIFY_EMAIL = 'supersekang@gmail.com'; // ← 알림 받을 메일 const DRIVE_FOLDER = '오더문의 첨부'; const ADMIN_KEY = 'yj2026'; // ← admin.html 접근키. 꼭 바꾸세요. /** 시트 컬럼 순서 = 아래 KEYS 순서 */ const KEYS = ['createdAt','ticket','brand','person','phone','items', 'styles','colors','sizes','perUnit','total','dueDate','targetPrice', 'material','detail','spec','sewing','packing','qc','inspect', 'fabric','assets','channel','etc','address','images','status']; const HEADERS = ['접수시각','접수번호','브랜드·회사명','담당자','연락처','품목', '스타일수','컬러수','사이즈수','컬러사이즈당','총수량','납기일','희망단가', '주원자재·부자재','디테일','사이즈스펙','봉제·가공','포장','QC기준','검사방식', '원단확정','보유자료','판매채널','기타','주소','이미지','상태']; /* ===================== 읽기 (admin.html) ===================== */ /** JSONP 방식이라 브라우저 보안(CORS)에 걸리지 않습니다. */ function doGet(e) { const p = e.parameter || {}; const cb = p.callback; let out; if (p.action === 'list') { out = (p.key === ADMIN_KEY) ? { ok: true, rows: readAll_() } : { ok: false, error: 'BAD_KEY' }; } else if (p.action === 'img') { out = (p.key === ADMIN_KEY) ? imgData_(p.id) : { ok: false, error: 'BAD_KEY' }; } else { out = { ok: true, msg: 'order endpoint v2' }; } const body = JSON.stringify(out); return cb ? Utilities.formatDate(v, 'Asia/Seoul', 'yyyy-MM-dd') : Utilities.formatDate(v, 'Asia/Seoul', 'yyyy-MM-dd HH:mm'); } o[k] = (v === '' || v === null) ? '' : String(v).replace(/^'/, ''); }); return o; }); return rows.reverse(); // 최신 건이 위로 } /* ===================== 쓰기 (접수 / 수정 / 삭제) ===================== */ function doPost(e) { try { const d = JSON.parse(e.postData.contents); if (d.action === 'update') return guard_(d, update_); if (d.action === 'delete') return guard_(d, remove_); return json_(create_(d)); } catch (err) { console.error(err); return json_({ ok: false, error: String(err) }); } } function guard_(d, fn) { if (d.key !== ADMIN_KEY) return json_({ ok: false, error: 'BAD_KEY' }); return json_(fn(d)); } /** 새 접수 저장 */ function create_(d) { const sh = sheet_(); const links = saveImages_(d.images || [], d.ticket); const row = KEYS.map(function (k) { if (k === 'createdAt') return new Date(); if (k === 'images') return links.join('\n'); if (k === 'status') return '접수'; if (k === 'phone') return "'" + (d.phone || ''); return d[k] === undefined ? '' : d[k]; }); sh.appendRow(row); notify_(d, links); return { ok: true, ticket: d.ticket }; } /** 접수번호로 찾아 지정한 항목만 수정 */ function update_(d) { const sh = sheet_(); const r = findRow_(sh, d.ticket); if (r < 0) return { ok: false, error: 'NOT_FOUND' }; Object.keys(d.fields || {}).forEach(function (k) { const i = KEYS.indexOf(k); if (i < 0 || k === 'createdAt' || k === 'ticket' || k === 'images') return; const v = (k === 'phone') ? "'" + d.fields[k] : d.fields[k]; sh.getRange(r, i + 1).setValue(v); }); return { ok: true }; } /** 접수번호 배열을 받아 삭제 (아래쪽부터 지워야 행이 안 밀립니다) */ function remove_(d) { const sh = sheet_(); const rows = (d.tickets || []).map(function (t) { return findRow_(sh, t); }) .filter(function (r) { return r > 0; }) .sort(function (a, b) { return b - a; }); rows.forEach(function (r) { sh.deleteRow(r); }); return { ok: true, deleted: rows.length }; } function findRow_(sh, ticket) { if (sh.getLastRow() < 2) return -1; const col = sh.getRange(2, 2, sh.getLastRow() - 1, 1).getValues(); for (let i = 0; i < col.length; i++) { if (String(col[i][0]) === String(ticket)) return i + 2; } return -1; } /* ===================== 유틸 ===================== */ function sheet_() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; const head = sh.getRange(1, 1, 1, HEADERS.length); if (String(head.getValues()[0][0]) !== HEADERS[0]) { head.setValues([HEADERS]).setFontWeight('bold'); sh.setFrozenRows(1); } return sh; } function saveImages_(images, ticket) { if (!images.length) return []; const folder = getFolder_(); return images.map(function (img, i) { const parts = img.dataUrl.split(','); const blob = Utilities.newBlob( Utilities.base64Decode(parts[1]), 'image/jpeg', ticket + '_' + (i + 1) + '.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1000'; }); } function getFolder_() { const it = DriveApp.getFoldersByName(DRIVE_FOLDER); return it.hasNext() ? it.next() : DriveApp.createFolder(DRIVE_FOLDER); } function notify_(d, links) { if (!NOTIFY_EMAIL || NOTIFY_EMAIL.indexOf('example.com') > -1) return; const body = '접수번호 ' + d.ticket + '\n' + '브랜드 ' + d.brand + '\n' + '담당자 ' + d.person + ' / ' + d.phone + '\n' + '품목 ' + d.items + '\n' + '수량 ' + d.total + '장 (' + d.styles + '×' + d.colors + '×' + d.sizes + '×' + d.perUnit + ')\n' + '납기 ' + d.dueDate + '\n' + '희망단가 ' + (d.targetPrice || '-') + '\n' + (links.length ?
